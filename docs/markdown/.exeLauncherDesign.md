Building a native .exe wrapper for Java applications with GitHub release checkers requires two 
components: a native C/C++/Rust launcher, and a Java background updater.

The cleanest architecture mimics how JetBrains does it: C++ executable simply launches the app,
while the Java code handles the GitHub API requests and user interface dialogs.

> The updater logic could also be separate from the Java, building it in Python, C, C++, or Rust
> would be better for a case like this.

### The Native Launcher (In `C++`, `launcher.cpp`):

This lightweight executable finds the bundled Java Runtime, configures JVM arguments, and loads
the JVM dynamically using the ***Java Native Interface (JNI)***

```cpp
    #include <windows.h>
    #include <jni.h>
    #include <string>
    
    typedef jint(JNICALL *CreateJavaVM_t)(JavaVM **, void **, void **);
    
    int WINAPI WinMain(HINSTANCE hInstance, HINSTANCE hPrevInstance, LPSTR lpCmdLine, int nCmdShow)
    {
        // Locate the JVM DLL (Assuming a bundled JRE in a "runtime" folder)
        std::wstring dllPath = L"runtime\\bin\\server\\jvm.dll";
        HMODULE hJvmDll = LoadLibraryW(dllPath.c_str());
        if (!hJvmDll)
        {
            MessageBoxW(NULL, L"Failed to load JVM.", L"Error", MB_ICONERROR);
            return 1;    
        }
    
        // Get the address of JNI_CreateJavaVM
        CreateJavaVM_t F_CreateJavaVM = (CreateJavaVM_t) GetProcAddress(hJvmDll, "JNI_CreateJavaVM");
        if (!F_CreateJavaVM)
        {
            return 1;    
        }
    
        // Set up JVM arguments (.vmoptions)
        JavaVMInitArgs vmArgs;
        JavaVMOptions options[2];
        
        // Manually chaining the dependencies using semicolons
        std::string classpath = "-Djava.class.path="
                        "ForgeIDE.jar;"
                        "lib/flatlaf-3.x.jar;"
                        "lib/jackson-core-2.x.jar;"
                        "lib/jackson-databind-2.x.jar;"
                        "lib/jackson-annotations-2.x.jar";
        
        options[1].optionString = (char*) "-Xmx512m"; // Memory limit
        
        vmArgs.version = JNI_VERSION_10;
        vmArgs.nOptions = 2;
        vmArgs.options = options;
        vmArgs.ignoreUnrecognised = JNI_TRUE;
        
        // Start the JVM
        if (F_CreateJavaVM(&jvm, (void **)&env, &vmArgs) < 0)
        {
            MessageBoxW(NULL, L"Failed to start JVM.", L"Error", MB_ICONERROR);
            return 1;    
        }
    
        // Find and invoke the Main Class
        jclass mainClass = env->FindClass("com/willclay/forgeide/Main");
        if (mainClass)
        {
            jmethodID mainMethod = env->GetStaticMethodID(mainClass, "main", "([Ljava/lang/String;)V");
            if (mainMethod)
            {
                jobjectArray args = env->NewObjectArray(0, env->FindClass("java/lang/String"), NULL);
                env->CallStaticVoidMethod(mainClass, mainMethod, args);    
            }
        }
    
        jvm->DestroyJavaVM();
        return 0;
    }
```

### The Java GitHub Update Checker

Running the update logic inside Java can give access to UI libraries (JavaFX or Swing) and simple
HTTP clients. 

Using Java's built-in HTTP client to query the public GitHub Releases API:

```java
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class UpdateChecker
{
    private static final String API_URL = "https://github.com";
    private static final String CURRENT_VERSION = "v1.0.0"; // Hardcoded or read from properties
    
    public static String getLatestVersion() 
    {
        try 
        {
            HttpClient client = HttpClient.newHttpClient();
            HttpRequest request = HttpRequest.newBuilder()
                    .uri(URI.create(API_URL))
                    .header("Accept", "application/vnd.github.v3+json")
                    .header("User-Agent", "Java-Update_Checker") // Github API requires a User-Agent
                    .GET()
                    .build();

            HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());

            if (response.statusCode() == 200) 
            {
                // Parse JSON to find the "tag_name" field
                // Simple parsing without external libraries for brevity:
                String body = response.body();
                int index = body.indexOf("\"tag_name\":");
                if (index != -1) 
                {
                    int start = body.indexOf("\"", index + 11) + 1;
                    int end = body.indexOf("\"", start);
                    return body.substring(start, end);
                }
            }
        } 
        catch (Exception e) 
        {
            System.err.println("Failed to check for updates: " + e.getMessage());
        }
        return CURRENT_VERSION;
    }
}
```

And then compare the versions when the application launches. If a mismatch occurs, show a standard
native-looking dialog window.

```java
import javax.swing.JOptionPane;
import javax.swing.SwingUtilities;

public class MainApp 
{
    public static void main(String[] args) 
    {
        // Run update check in a background thread to prevent startup lag
        Thread updateThread = new Thread(() -> 
        {
            String latestVersion = UpdateChecker.getLatestVersionTag();
            if (!latestVersion.equals(UpdateChecker.CURRENT_VERSION)) 
            {
                
                // Switch back to UI thread to show dialog
                SwingUtilities.invokeLater(() -> 
                {
                    int response = JOptionPane.showConfirmDialog(
                            null,
                            "A new version (" + latestVersion + ") is available! Would you like to download it?",
                            "Update Available",
                            JOptionPane.YES_NO_OPTION,
                            JOptionPane.INFORMATION_MESSAGE
                    );
                    
                    if (response == JOptionPane.YES_OPTION) 
                    {
                        // Redirect user to GitHub releases page
                        try 
                        {
                            java.awt.Desktop.getDesktop().browse(new java.net.URI("https://github.com"));
                        } 
                        catch (Exception e) 
                        {
                            e.printStackTrace();
                        }
                    }
                });
            }
        });
        
        updateThread.setDaemon(true);
        updateThread.start();

        // Initialize your main application UI here...
    }
}
```
